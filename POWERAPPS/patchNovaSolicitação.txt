If(
    IsBlank(CBoxCavalo.Selected.Value) ||
    IsBlank(CBoxDestino.Selected.Value) ||
    IsBlank(CBoxCPF.Selected.Value) ||
    IsBlank(CBoxMotorista.Selected.Value) ||
    IsBlank(CBoxCodigo.Selected.Value) ||
    IsBlank(CBoxTransportadora.Selected.Value) ||
    IsBlank(CBoxMotivo.Selected.Value) ||

    !(
        ToggleEmissãoNota_2.Value ||
        ToggleTrocaGuia_2.Value ||
        ToggleTrocaCarreta_2.Value ||
        ToggleCancelarGuia_2.Value
    );

    /* ERRO DE VALIDAÇÃO */

    Set(varMostrarErroTipoSolicitacao; false);;
    Set(varMostrarErroTipoSolicitacao; true);;
    Set(varMostrarErros; true);;

    Notify(
        "Existem campos obrigatórios não preenchidos. Os campos destacados precisam de atenção.";
        NotificationType.Warning;
        5000
    );

    /* VALIDAÇÃO DO CADASTRO DO VEÍCULO */

    If(
        IsBlank(varUltimaViagem.ID) ||
        varUltimaViagem.Cavalo <> CBoxCavalo.Selected.Value;
    
        Notify(
            "Selecione um veículo cadastrado pelo GO ou pela placa. Se ele não existir, faça o cadastro antes de salvar.";
            NotificationType.Warning;
            5000
        );
    
        /* SALVAMENTO */

        IfError(
            Patch(
                solicitacoesViagens;
                Defaults(solicitacoesViagens);
                {
                    Title:
                        CBoxCavalo.Selected.Value & " - " &
                        Text(Now(); "yyyymmddhhmmss");

                    GO: If(
                            IsBlank(varUltimaViagem.GO);
                            Blank();
                            varUltimaViagem.GO
                        );
                    Cavalo: CBoxCavalo.Selected.Value;
                    Reboque1: CBoxReboque1.Selected.Value;
                    Reboque2: CBoxReboque2.Selected.Value;
                    Composicao: CBoxComposição.Selected.Value;
                    CPF: CBoxCPF.Selected.Value;
                    Motorista: CBoxMotorista.Selected.Value;

                    CodigoTransportadora:
                        Value(CBoxCodigo.Selected.Value);

                    Transportadora:
                        CBoxTransportadora.Selected.Value;

                    Destino: CBoxDestino.Selected.Value;
                    Motivo: {Value: CBoxMotivo.Selected.Value};
                    Observacoes: TTMultObservacoes.Text;
                    EmissaoGuia: ToggleEmissãoNota_2.Value;
                    TrocaGuia: ToggleTrocaGuia_2.Value;
                    TrocaCarreta: ToggleTrocaCarreta_2.Value;
                    EnviarPDF: ToggleEnviarPDF_2.Value;
                    CancelarGuia: ToggleCancelarGuia_2.Value;
                    Status: {Value: CBoxStatus.Selected.Value}
                }
            );; true;

            /* FALHA: MANTÉM OS DADOS NA TELA */

            Notify(
                "Não foi possível salvar a solicitação: " &
                FirstError.Message;
                NotificationType.Error;
                5000
            );; false;

            /* SUCESSO: LIMPA VARIÁVEIS E CONTROLES */

            Set(varUltimaViagem; Blank());;
            Set(varMotoristaSolicitacao; Blank());;

            Reset(CBoxGO);;
            Reset(CBoxCavalo);;
            Reset(CBoxDestino);;
            Reset(CBoxCPF);;
            Reset(CBoxMotorista);;
            Reset(CBoxCodigo);;
            Reset(CBoxTransportadora);;
            Reset(CBoxMotivo);;
            Reset(TTMultObservacoes);;
            Reset(CBoxReboque1);;
            Reset(CBoxReboque2);;
            Reset(CBoxComposição);;

            Reset(ToggleEmissãoNota_2);;
            Reset(ToggleTrocaGuia_2);;
            Reset(ToggleTrocaCarreta_2);;
            Reset(ToggleEnviarPDF_2);;
            Reset(ToggleCancelarGuia_2);;

            Set(varMostrarErros; false);;
            Set(varMostrarErroTipoSolicitacao; false);;

            Refresh(solicitacoesViagens);;

            Notify(
                "Solicitação criada com sucesso!";
                NotificationType.Success;
                5000
            );; true
        )
    )
)
